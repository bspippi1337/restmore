cmake_minimum_required(VERSION 3.21)
project(restmore VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(RESTMORE_NETWORK "Build HTTPS documentation crawler with libcurl" ON)
option(RESTMORE_GUI "Build native raylib UI (all application logic remains C)" OFF)
option(RESTMORE_SANITIZE "Address/undefined behavior sanitizers (Clang/GCC)" OFF)
option(RESTMORE_TESTS "Build pure C regression tests" ON)
if(RESTMORE_NETWORK)
  find_package(CURL REQUIRED)
endif()
add_library(restmore STATIC
  src/core.c
  src/sha256.c
  src/json.c
  src/catalog.c
  src/exports.c
  src/verification.c
)
target_include_directories(restmore PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
target_compile_definitions(restmore PRIVATE _POSIX_C_SOURCE=200809L)
target_compile_definitions(restmore PUBLIC RESTMORE_NETWORK_ENABLED=$<BOOL:${RESTMORE_NETWORK}>)
if(ANDROID)
  set_target_properties(restmore PROPERTIES POSITION_INDEPENDENT_CODE ON)
endif()
if(RESTMORE_NETWORK)
  target_sources(restmore PRIVATE src/network.c)
  target_link_libraries(restmore PUBLIC CURL::libcurl)
  # The bounded HTTPS resolver uses Winsock directly for address validation.
  # MSVC does not link Winsock automatically when linking libcurl.
  if(WIN32)
    target_link_libraries(restmore PUBLIC ws2_32)
  endif()
else()
  target_sources(restmore PRIVATE src/network_off.c)
endif()
if(MSVC)
  target_compile_options(restmore PRIVATE /W4)
else()
  target_compile_options(restmore PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
if(RESTMORE_SANITIZE AND CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
if(NOT ANDROID AND NOT IOS)
  add_executable(restmore-cli src/cli.c)
  target_link_libraries(restmore-cli PRIVATE restmore)
endif()
if(RESTMORE_GUI)
  if(ANDROID)
    # raylib builds the EGL/OpenGL ES NativeActivity backend for Android.
    set(PLATFORM Android CACHE STRING "raylib target platform" FORCE)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "Bundle raylib into app .so" FORCE)
  endif()
  include(FetchContent)
  set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(raylib GIT_REPOSITORY https://github.com/raysan5/raylib.git GIT_TAG 5.5 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(raylib)
  if(ANDROID)
    add_library(restmore-gui SHARED src/gui.c)
  else()
    add_executable(restmore-gui src/gui.c)
  endif()
  target_link_libraries(restmore-gui PRIVATE restmore raylib)
  if(ANDROID)
    target_link_libraries(restmore-gui PRIVATE android log EGL GLESv2 OpenSLES m)
    # NativeActivity discovers this symbol through android_native_app_glue.
    target_link_options(restmore-gui PRIVATE -Wl,-u,ANativeActivity_onCreate)
  endif()
  if(NOT WIN32)
    find_package(Threads REQUIRED)
    target_link_libraries(restmore-gui PRIVATE Threads::Threads)
  endif()
endif()
if(RESTMORE_TESTS AND NOT CMAKE_CROSSCOMPILING)
  enable_testing()
  add_executable(restmore-tests tests/test_core.c)
  target_link_libraries(restmore-tests PRIVATE restmore)
  add_test(NAME C11_core_sha256_dns_openapi_export COMMAND restmore-tests)
  add_executable(restmore-verify-tests tests/test_verification.c)
  target_link_libraries(restmore-verify-tests PRIVATE restmore)
  add_test(NAME C11_verification_human_authorization_and_contract COMMAND restmore-verify-tests)
endif()
include(GNUInstallDirs)
install(TARGETS restmore ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/restmore DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
if(TARGET restmore-cli)
  install(TARGETS restmore-cli RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
if(TARGET restmore-gui)
  install(TARGETS restmore-gui RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
